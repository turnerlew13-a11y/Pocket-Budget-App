Dedicated Year Sheet Setup

1 Sheet per Year generator script

Step-by-step setup to create your annual sheet & connect it with Pocket Budget:

  1. Create a blank Google Sheet (e.g. named "Pocket Budget 2026").
  2. Click Extensions > Apps Script at the top of the Google Sheet.
  3. Delete any existing code in the editor, and paste the entire code block below.
  4. Select the buildYearSpreadsheet function from the dropdown toolbar and click Run (grant permissions once). This creates all 12 monthly tabs formatted exactly with Expenses on the left and Categories on the right, plus the Annual Dashboard tab!
  5. Click the blue Deploy > New deployment button in the top right.
  6. Select Web app. Set Execute as: Me and Who has access: Anyone.
  7. Copy the resulting Web app URL and paste it into Pocket Budget under Settings > Google Sheets!
// --- POCKET BUDGET REAL-TIME HOUSEHOLD SYNC SCRIPT ---
const TARGET_YEAR = 2026;

function buildYearSpreadsheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const months = ["January","February","March","April","May","June","July","August","September","October","November","December"];
  
  // 1. Master Transactions Ledger Sheet (Used for syncing between household devices)
  let txMaster = ss.getSheetByName("Transactions") || ss.insertSheet("Transactions");
  if (txMaster.getLastRow() === 0) {
    txMaster.appendRow(["ID", "Date", "Amount", "Category", "Subcategory", "PaymentMethod", "Type", "Timestamp"]);
    txMaster.getRange("A1:H1").setFontWeight("bold").setBackground("#4f46e5").setFontColor("#ffffff");
  }

  // 2. Household Metadata Tab (for categories and budget rules)
  let metaSheet = ss.getSheetByName("Household_Meta") || ss.insertSheet("Household_Meta");
  if (metaSheet.getLastRow() === 0) {
    metaSheet.appendRow(["Key", "JSON_Data"]);
    metaSheet.getRange("A1:B1").setFontWeight("bold").setBackground("#334155").setFontColor("#ffffff");
  }

  // 3. Annual Dashboard Tab
  let dash = ss.getSheetByName("Annual Dashboard") || ss.insertSheet("Annual Dashboard", 0);
  dash.clear();
  dash.getRange("A1").setValue("POCKET BUDGET - " + TARGET_YEAR + " ANNUAL DASHBOARD").setFontWeight("bold").setFontSize(14).setFontColor("#4f46e5");
  dash.getRange("A3:D3").setValues([["Month", "Total Income ($)", "Total Spent ($)", "Net Gain/Loss ($)"]]).setFontWeight("bold").setBackground("#f1f5f9");
  
  for (let i = 0; i < 12; i++) {
    const row = 4 + i;
    const mName = months[i];
    dash.getRange(row, 1).setValue(mName);
    dash.getRange(row, 2).setFormula(`=IFERROR(SUMIFS(Transactions!C:C, Transactions!G:G, "income", Transactions!B:B, ">="&"${TARGET_YEAR}-"&TEXT(${i+1},"00")&"-01", Transactions!B:B, "<="&"${TARGET_YEAR}-"&TEXT(${i+1},"00")&"-31"), 0)`);
    dash.getRange(row, 3).setFormula(`=IFERROR(SUMIFS(Transactions!C:C, Transactions!G:G, "expense", Transactions!B:B, ">="&"${TARGET_YEAR}-"&TEXT(${i+1},"00")&"-01", Transactions!B:B, "<="&"${TARGET_YEAR}-"&TEXT(${i+1},"00")&"-31"), 0)`);
    dash.getRange(row, 4).setFormula(`=B${row}-C${row}`);
  }

  // 4. Monthly Formatted Sheets
  months.forEach((m) => {
    let mSheet = ss.getSheetByName(m) || ss.insertSheet(m);
    if (mSheet.getLastRow() === 0) {
      mSheet.getRange("A1:D1").setValues([["Date", "Category", "$ Amount", "Payment Method"]]).setFontWeight("bold").setBackground("#dbeafe").setFontColor("#1e3a8a");
      mSheet.getRange("H1:K1").setValues([["Expense Category", "", "Budget", "Actual"]]).setFontWeight("bold").setBackground("#dbeafe").setFontColor("#1e3a8a");
      mSheet.getRange("H18:K18").setValues([["Total", "", "=SUM(J2:J17)", "=SUM(K2:K17)"]]).setFontWeight("bold");
      mSheet.getRange("H20:K21").setValues([
        ["Net Gain/Loss", "", "Budget", "Actual"],
        ["", "", "=O18-J18", "=P18-K18"]
      ]).setFontWeight("bold");
      mSheet.getRange("H20").setBackground("#bfdbfe");
      mSheet.getRange("N1:P1").setValues([["Income Category", "Budget", "Actual"]]).setFontWeight("bold").setBackground("#dcfce7").setFontColor("#14532d");
      mSheet.getRange("N18:P18").setValues([["Total", "=SUM(O2:O17)", "=SUM(P2:P17)"]]).setFontWeight("bold");
    }
  });

  SpreadsheetApp.flush();
}

function doPost(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const payload = JSON.parse(e.postData.contents);
    
    // Check if bulk sync or single transaction
    if (payload.action === "sync_all") {
      let metaSheet = ss.getSheetByName("Household_Meta") || ss.insertSheet("Household_Meta");
      metaSheet.clear();
      metaSheet.appendRow(["Key", "JSON_Data"]);
      metaSheet.appendRow(["categories", JSON.stringify({
        expenseCategories: payload.expenseCategories || [],
        incomeCategories: payload.incomeCategories || [],
        subcategories: payload.subcategories || {},
        subcategoryRules: payload.subcategoryRules || {},
        monthlyBudgetLedger: payload.monthlyBudgetLedger || {}
      })]);

      let txSheet = ss.getSheetByName("Transactions") || ss.insertSheet("Transactions");
      txSheet.clear();
      txSheet.appendRow(["ID", "Date", "Amount", "Category", "Subcategory", "PaymentMethod", "Type", "Timestamp"]);
      
      const txs = payload.transactions || [];
      const rows = txs.map(t => [t.id, t.date, Number(t.amount), t.category, t.subcategory || "", t.paymentMethod, t.type, new Date()]);
      if (rows.length > 0) {
        txSheet.getRange(2, 1, rows.length, 8).setValues(rows);
      }
      return ContentService.createTextOutput(JSON.stringify({ status: "success", count: txs.length })).setMimeType(ContentService.MimeType.JSON);
    }

    // Default: Single transaction push
    let txSheet = ss.getSheetByName("Transactions") || ss.insertSheet("Transactions");
    txSheet.appendRow([payload.id || Date.now().toString(), payload.date, Number(payload.amount), payload.purpose, payload.subcat || "", payload.how, payload.type, new Date()]);

    // Also append to the corresponding monthly visual tab
    const dateObj = new Date(payload.date + 'T00:00:00');
    const months = ["January","February","March","April","May","June","July","August","September","October","November","December"];
    const monthName = months[dateObj.getMonth()];
    let mSheet = ss.getSheetByName(monthName);
    if (mSheet) {
      if (payload.type === 'expense') {
        let r = 2;
        while (mSheet.getRange(r, 1).getValue() !== "" && r < 500) r++;
        mSheet.getRange(r, 1, 1, 4).setValues([[payload.date, payload.purpose, Number(payload.amount), payload.how]]);
      } else {
        let r = 2;
        while (mSheet.getRange(r, 14).getValue() !== "" && r < 18) {
          if (mSheet.getRange(r, 14).getValue() === payload.purpose) break;
          r++;
        }
        mSheet.getRange(r, 14).setValue(payload.purpose);
        let curr = Number(mSheet.getRange(r, 16).getValue()) || 0;
        mSheet.getRange(r, 16).setValue(curr + Number(payload.amount));
      }
    }

    return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let txSheet = ss.getSheetByName("Transactions");
    let transactions = [];
    if (txSheet && txSheet.getLastRow() > 1) {
      const rows = txSheet.getRange(2, 1, txSheet.getLastRow() - 1, 8).getValues();
      transactions = rows.map(r => {
        let dt = r[1] instanceof Date ? Utilities.formatDate(r[1], Session.getScriptTimeZone(), "yyyy-MM-dd") : String(r[1]).split('T')[0];
        return {
          id: String(r[0] || Date.now()),
          date: dt,
          amount: parseFloat(r[2]) || 0,
          category: String(r[3] || "General"),
          subcategory: String(r[4] || ""),
          paymentMethod: String(r[5] || "Debit"),
          type: String(r[6] || "expense").toLowerCase()
        };
      });
    }

    let meta = {};
    let metaSheet = ss.getSheetByName("Household_Meta");
    if (metaSheet && metaSheet.getLastRow() > 1) {
      try {
        const rawJson = metaSheet.getRange(2, 2).getValue();
        if (rawJson) meta = JSON.parse(rawJson);
      } catch (err) {}
    }

    return ContentService.createTextOutput(JSON.stringify({
      transactions: transactions,
      meta: meta
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ error: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}
